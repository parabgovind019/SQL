(select 
requester_id as id,
count(requester_id) as num
from RequestAccepted group by requester_id)
union all
(select 
r1.accepter_id as id,
count(r1.accepter_id) +

(select count(requester_id) from RequestAccepted where r1.accepter_id = requester_id),0) as num
from RequestAccepted r1 group by r1.accepter_id))
select cte.id, max(cte.num) as num from cte group by cte.id order by num desc limit 1;
