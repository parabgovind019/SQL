delimiter $$
create procedure if not exists prime(out primeList text)
begin 
declare n int default 2;
declare m int default 2;
loop1: while n <= 1000 do 
    loop2: while m <= sqrt(n) do 
        if n % m = 0 then 
            set n = n+1;
            set m = 2;
            iterate loop1;
        end if;
        set m = m+1;
    end while;
    set primeList = concat_ws('&',primeList,n);
    set m = 2;
    set n = n+1;
end while;
end $$
delimiter ;

call prime(@list);
select @list;

https://www.hackerrank.com/challenges/full-score/problem?isFullScreen=true

SELECT
    h.hacker_id,
    h.name
FROM Submissions s
JOIN Challenges c
    ON s.challenge_id = c.challenge_id
JOIN Difficulty d
    ON c.difficulty_level = d.difficulty_level
JOIN Hackers h
    ON s.hacker_id = h.hacker_id
WHERE s.score = d.score
GROUP BY h.hacker_id, h.name
HAVING COUNT(DISTINCT s.challenge_id) > 1
ORDER BY COUNT(DISTINCT s.challenge_id) desc, h.hacker_id;

WITH all_record AS (
    SELECT 
        s.ID,
        s.Name,
        p.Salary AS his_sal,
        f.Friend_ID,
        pf.Salary AS frd_sal
    FROM Students s
    JOIN Friends f 
        ON s.ID = f.ID
    JOIN Packages p 
        ON s.ID = p.ID
    JOIN Packages pf 
        ON f.Friend_ID = pf.ID
)
SELECT Name
FROM all_record
WHERE his_sal < frd_sal
ORDER BY frd_sal;

SELECT h.hacker_id, h.name, COUNT(c.challenge_id) AS total
FROM hackers h
JOIN challenges c ON h.hacker_id = c.hacker_id
GROUP BY h.hacker_id, h.name
HAVING total = (
        SELECT MAX(cnt)
        FROM (
            SELECT COUNT(*) AS cnt
            FROM challenges
            GROUP BY hacker_id
        ) m
     )
   OR total NOT IN (
        SELECT cnt
        FROM (
            SELECT COUNT(*) AS cnt
            FROM challenges
            GROUP BY hacker_id
        ) d
        GROUP BY cnt
        HAVING COUNT(*) > 1
     )
ORDER BY total DESC, h.hacker_id;
